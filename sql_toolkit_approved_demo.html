using System; using System.IO; using System.Text.Json; using System.Windows.Forms; using Microsoft.Web.WebView2.WinForms; using Microsoft.Web.WebView2.Core; // שים לב: אם ה-namespace כאן שונה מהשם שנתת לפרויקט ב-Visual Studio, // שנה את המילה "AutoReportOut" בשורה הבאה כך שתתאים לשם הפרויקט שלך. namespace AutoReportOut { public class Settings { public string Url { get; set; } = ""; public string ButtonSelector { get; set; } = "#ctl00_ctl00_uxContentPlaceHolder_uxContentPlaceHolder_uxReportOutTime"; public string TargetTime { get; set; } = "16:00:00"; public int KeepAliveMinutes { get; set; } = 4; } public partial class Form1 : Form { private WebView2 webView; private System.Windows.Forms.Timer timer; private Label statusLabel; private Settings settings; private DateTime lastKeepAlive; private bool alreadyClicked = false; public Form1() { InitializeComponent(); // נוצר אוטומטית ע"י Visual Studio, אל תמחק this.Text = "דיווח יציאה אוטומטי"; this.Width = 1000; this.Height = 700; statusLabel = new Label(); statusLabel.Dock = DockStyle.Top; statusLabel.Height = 30; statusLabel.Text = "טוען..."; statusLabel.TextAlign = System.Drawing.ContentAlignment.MiddleCenter; this.Controls.Add(statusLabel); webView = new WebView2(); webView.Dock = DockStyle.Fill; this.Controls.Add(webView); LoadSettings(); InitWebView(); timer = new System.Windows.Forms.Timer(); timer.Interval = 1000; timer.Tick += Timer_Tick; } private void LoadSettings() { string path = Path.Combine(AppDomain.CurrentDomain.BaseDirectory, "settings.json"); if (File.Exists(path)) { string json = File.ReadAllText(path); settings = JsonSerializer.Deserialize(json) ?? new Settings(); } else { settings = new Settings(); File.WriteAllText(path, JsonSerializer.Serialize(settings, new JsonSerializerOptions { WriteIndented = true })); } } private async void InitWebView() { string profileFolder = Path.Combine(AppDomain.CurrentDomain.BaseDirectory, "BrowserProfile"); var env = await CoreWebView2Environment.CreateAsync(userDataFolder: profileFolder); await webView.EnsureCoreWebView2Async(env); webView.CoreWebView2.Navigate(settings.Url); lastKeepAlive = DateTime.Now; timer.Start(); } private async void Timer_Tick(object sender, EventArgs e) { DateTime now = DateTime.Now; // רענון אמיתי כל X דקות - זו הדרך הכי אמינה למנוע ניתוק, // כי היא "מאתחלת" את הדף בדיוק כמו לחיצה על F5. // חשוב: זה לא הורס את הקוד שלנו, כי הוא רץ כאן ב-C#, לא בתוך הדף עצמו. if ((now - lastKeepAlive).TotalMinutes >= settings.KeepAliveMinutes) { lastKeepAlive = now; try { webView.CoreWebView2.Reload(); } catch { // אם זה נכשל פעם אחת זה לא סוף העולם, ננסה שוב בפעם הבאה } } TimeSpan targetTimeOfDay; bool parsed = TimeSpan.TryParse(settings.TargetTime, out targetTimeOfDay); statusLabel.Text = $"השעה עכשיו: {now:HH:mm:ss} | שעת יעד: {settings.TargetTime} | " + (alreadyClicked ? "בוצעה לחיצה ✔" : "ממתין..."); if (parsed && !alreadyClicked && now.TimeOfDay >= targetTimeOfDay) { alreadyClicked = true; try { string script = $"var btn = document.querySelector('{settings.ButtonSelector}'); if(btn) {{ btn.click(); 'clicked'; }} else {{ 'not-found'; }}"; string result = await webView.CoreWebView2.ExecuteScriptAsync(script); statusLabel.Text = "בוצעה לחיצה ✔ (תוצאה: " + result + ") - אפשר לסגור את החלון"; } catch (Exception ex) { statusLabel.Text = "שגיאה בלחיצה: " + ex.Message; } } } } }
